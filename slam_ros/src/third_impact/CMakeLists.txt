cmake_minimum_required(VERSION 3.8)
project(third_impact)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(dwb_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
add_library(third_impact_panel SHARED cpp/goal_panel.cpp include/third_impact/goal_panel.hpp)
target_include_directories(third_impact_panel PRIVATE include)
target_link_libraries(third_impact_panel Qt5::Widgets)
ament_target_dependencies(third_impact_panel rclcpp std_msgs rviz_common pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS third_impact_panel LIBRARY DESTINATION lib)
add_library(third_impact_critics SHARED cpp/logged_oscillation.cpp)
ament_target_dependencies(third_impact_critics dwb_core nav2_util rclcpp rclcpp_lifecycle std_msgs pluginlib)
pluginlib_export_plugin_description_file(dwb_core critic_plugins.xml)
install(TARGETS third_impact_critics LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/impact_node scripts/debug_recorder scripts/localization_bootstrap scripts/amcl_scan DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz maps DESTINATION share/${PROJECT_NAME})
ament_package()
